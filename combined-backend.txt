const mongoose = require('mongoose');
const bcrypt = require('bcryptjs');

const adminSchema = new mongoose.Schema(
  {
    firstName: { type: String, required: true, trim: true },
    lastName: { type: String, required: true, trim: true },
    email: { type: String, required: true, unique: true, lowercase: true, trim: true },
    password: { type: String, required: true, minlength: 6 },
    phone: { type: String, default: '' },
    photo: { type: String, default: '' },
    role: { type: String, default: 'admin', immutable: true },
    // Password reset
    resetTokenHash: { type: String, default: '' },
    resetTokenExpires: { type: Date, default: null },
  },
  { timestamps: true }
);

adminSchema.pre('save', async function (next) {
  if (!this.isModified('password')) return next();
  this.password = await bcrypt.hash(this.password, 10);
  next();
});

adminSchema.methods.matchPassword = function (entered) {
  return bcrypt.compare(entered, this.password);
};

module.exports = mongoose.model('Admin', adminSchema);

models/Exam.js
const mongoose = require('mongoose');
const { MultiLangString } = require('./_shared');

/**
 * One question in an exam. Supports MCQ, true/false, matching, fill-blanks,
 * ordering, and short-answer. Fields not relevant to the active type are
 * simply ignored.
 *
 *   mcq          -> options[], correctIndices[]
 *   truefalse    -> correctBoolean
 *   matching     -> matchPairs[{ left, right }] (server presents shuffled, scores by pair)
 *   fillblank    -> blanks[{ acceptable: [str] }]   (per-blank acceptable answers, per language)
 *   ordering     -> items[] in correct order; client shuffles
 *   shortanswer  -> acceptableAnswers[] (per language; manual review possible too)
 */
const questionSchema = new mongoose.Schema(
  {
    type: {
      type: String,
      enum: ['mcq', 'truefalse', 'matching', 'fillblank', 'ordering', 'shortanswer'],
      required: true,
    },
    question: { type: MultiLangString, required: true },
    score: { type: Number, default: 1 },

    // MCQ
    options: [{ type: MultiLangString }],
    correctIndices: [{ type: Number }], // supports single or multi-correct

    // True / False
    correctBoolean: { type: Boolean },

    // Matching
    matchPairs: [
      {
        left: { type: MultiLangString },
        right: { type: MultiLangString },
      },
    ],

    // Fill in the blanks: question text contains "___" placeholders;
    // `blanks[i].acceptable` is the list of acceptable values for the i-th blank.
    blanks: [
      {
        acceptable: [{ type: String }], // case-insensitive compare on the server
      },
    ],

    // Ordering: `items` are stored in the correct order.
    items: [{ type: MultiLangString }],

    // Short answer: acceptable answers (also a free-text record is kept for review)
    acceptableAnswers: [{ type: String }],
  },
  { _id: true }
);

const examSchema = new mongoose.Schema(
  {
    title: { type: MultiLangString, required: true },
    section: { type: String, required: true },
    level: { type: String, enum: ['A', 'B', 'C', 'D'], required: true },

    // 'lessons' = covers a subset of lessons in a unit
    // 'unit'    = final exam for a whole unit (passing it unlocks the next unit)
    type: { type: String, enum: ['lessons', 'unit'], required: true },

    // Linked content
    lessons: [{ type: mongoose.Schema.Types.ObjectId, ref: 'Lesson' }],
    unit: { type: mongoose.Schema.Types.ObjectId },           // logical unit id, used for unlock
    unitName: { type: MultiLangString, default: () => ({}) },

    teacher: { type: mongoose.Schema.Types.ObjectId, ref: 'Teacher' },

    questions: [questionSchema],
    passingGrade: { type: Number, default: 50 }, // percentage
  },
  { timestamps: true }
);

examSchema.index({ level: 1, section: 1 });

module.exports = mongoose.model('Exam', examSchema);

models/Homework.js
const mongoose = require('mongoose');
const { MultiLangString } = require('./_shared');

/**
 * One prompt inside a writing homework. The teacher writes the partial
 * fragment (e.g. `بـ ـ ـ ـ`) in the multi-language `prompt` field, and a
 * single canonical `answer` string the student is supposed to type. The
 * student's answer is graded with a simple normalised case-insensitive
 * compare on the server (Arabic doesn't have case, but we still trim and
 * collapse whitespace so trailing diacritics don't break a match).
 */
const writingPromptSchema = new mongoose.Schema(
  {
    prompt: { type: MultiLangString, required: true },
    answer: { type: String, required: true, trim: true },
  },
  { _id: false }
);

/**
 * Homework. One document per assignment.
 *
 *   kind = 'writing'           -> uses `prompts[]`
 *   kind = 'recordedReading'   -> student uploads an audio recording
 *   kind = 'bookReading'       -> student reads a physical page; parent
 *                                 leaves a note in the submission
 *
 * `level` scopes the homework to all students at that level. Optionally,
 * `assignedTo` narrows it to a specific subset of students. An empty
 * `assignedTo` means "everyone at this level".
 */
const homeworkSchema = new mongoose.Schema(
  {
    kind: {
      type: String,
      enum: ['writing', 'recordedReading', 'bookReading'],
      required: true,
    },
    title: { type: MultiLangString, required: true },
    instructions: { type: MultiLangString, default: () => ({}) },
    level: { type: String, enum: ['A', 'B', 'C', 'D'], required: true },
    assignedTo: [{ type: mongoose.Schema.Types.ObjectId, ref: 'Student' }],
    teacher: { type: mongoose.Schema.Types.ObjectId, ref: 'Teacher' },
    dueDate: { type: Date, default: null },
    prompts: { type: [writingPromptSchema], default: undefined }, // writing only
  },
  { timestamps: true }
);

homeworkSchema.index({ level: 1, dueDate: 1 });

module.exports = mongoose.model('Homework', homeworkSchema);

models/TeacherAttendance.js
const mongoose = require('mongoose');

/**
 * One record per (teacher, day). The `hours` array holds the integer
 * hour-numbers the teacher actually worked that day — the spec uses
 * "1st h / 2nd h / 3rd h" but we keep it as a flexible array of integers
 * so a fourth (or fifth) hour can be added later without a migration.
 *
 * Saving for a (teacher, day) that already has a record overwrites it, so
 * the admin can revise without creating duplicates. That's enforced by the
 * bulkUpsert path on the controller.
 */
const teacherAttendanceSchema = new mongoose.Schema(
  {
    teacher: { type: mongoose.Schema.Types.ObjectId, ref: 'Teacher', required: true },
    date: { type: Date, required: true },
    hours: {
      type: [Number],
      default: [],
      validate: {
        validator: (arr) => arr.every((n) => Number.isInteger(n) && n >= 1 && n <= 12),
        message: 'hours must be integers between 1 and 12',
      },
    },
    note: { type: String, default: '' },
    recordedBy: { type: mongoose.Schema.Types.ObjectId, ref: 'Admin' },
  },
  { timestamps: true }
);

// Quick lookups by date and per-teacher.
teacherAttendanceSchema.index({ date: 1 });
teacherAttendanceSchema.index({ teacher: 1, date: -1 });
// One record per (teacher, day). The bulkWrite uses upsert against this
// pair so the unique index isn't strictly required, but it's cheap insurance.
teacherAttendanceSchema.index({ teacher: 1, date: 1 }, { unique: true });

module.exports = mongoose.model('TeacherAttendance', teacherAttendanceSchema);


models/_shared.js
const { Schema } = require('mongoose');

/**
 * Multi-language string. Every user-facing text in the DB uses this shape:
 *   { en: '...', ar: '...', nl: '...' }
 */
const MultiLangString = new Schema(
  {
    en: { type: String, default: '' },
    ar: { type: String, default: '' },
    nl: { type: String, default: '' },
  },
  { _id: false }
);

/**
 * Single icon entry inside an `imageWithIcons` block. Each icon has its own
 * image (typically a small PNG/SVG of a boy / girl / man) and its own audio
 * which plays when the icon is tapped.
 */
const IconEntry = new Schema(
  {
    kind: { type: String, enum: ['boy', 'girl', 'man'], required: true },
    iconUrl: { type: String, default: '' },   // small image displayed as the icon
    audioUrl: { type: String, default: '' },  // audio that plays on tap
    label: { type: MultiLangString, default: () => ({}) }, // optional caption per icon
  },
  { _id: false }
);

/**
 * Flexible content block. Used by Lesson and News.
 *
 *   text           { content: MultiLangString }
 *   image          { url, size }
 *   video          { url, size }
 *   imageAudio     { url, audioUrl, size }
 *   imageWithIcons { url, icons: [{kind, iconUrl, audioUrl, label}], size }
 *
 * `size` is one of '25' | '50' | '75' | '100' (percent of the row). The
 * lesson/news viewer puts blocks inside a flex-wrap row so e.g. four 25%
 * blocks land on the same line.
 */
const ContentBlock = new Schema(
  {
    type: {
      type: String,
      enum: ['text', 'image', 'video', 'imageAudio', 'imageWithIcons'],
      required: true,
    },
    order: { type: Number, default: 0 },
    size: { type: String, enum: ['25', '50', '75', '100'], default: '100' },
    // text
    content: { type: MultiLangString, default: () => ({}) },
    // media
    url: { type: String, default: '' },
    audioUrl: { type: String, default: '' },
    // imageWithIcons
    icons: { type: [IconEntry], default: undefined },
    // optional caption (multi-lang)
    caption: { type: MultiLangString, default: () => ({}) },
  },
  { _id: true }
);

 * lesson images/videos/audio, and note attachments through this endpoint.

    size: req.file.size,
    originalname: req.file.originalname,
  });
});

utils/generatePassword.js
/**
 * Auto-generate a password from a person's name following the project rule:
 *   [firstName][lastName]salahaddin   (lowercase, no spaces)
 *
 * Example: ('John', 'Doe Smith') -> 'johndoesmithsalahaddin'
 */
function generatePassword(firstName, lastName) {
  const clean = (s) => String(s || '').toLowerCase().replace(/\s+/g, '');
  return `${clean(firstName)}${clean(lastName)}salahaddin`;
}

utils/generatePayslipPDF.js
const PDFDocument = require('pdfkit');
const fs = require('fs');
const path = require('path');

/**
 * Minimalist payslip layout (one page, A4 portrait, 50pt margins):
 *
 *   ISSUED TO                    PAY PERIOD
 *   Name                         From:    DD/MM/YYYY
 *   Email                        To:      DD/MM/YYYY
 *   Phone                        Issued:  DD/MM/YYYY
 *   ────────────────────────────────────────────
 *   EARNINGS
 *   Description       Hours    Rate     Amount
 *   ...rows...
 *   ────────────────────────────────────────────
 *                            ┌─────────────┐
 *                            │ TOTAL DUE   │
 *                            │ € XX.XX     │
 *                            └─────────────┘
 *
 *   NOTES (if any)
 *
 *   Authorised signature       Teacher signature
 *   ___________________        ___________________
 *
 * No brand header, reference, or footer. Sections flow naturally with a
 * tracked Y cursor so the page never overflows.
 */

const PALETTE = {
  primary:    '#0D274D',
  primaryFg:  '#FFFFFF',
  accent:     '#C9A054',
  ink:        '#111827',
  body:       '#374151',
  muted:      '#6B7280',
  divider:    '#E5E7EB',
  zebra:      '#F8FAFC',
};

function generatePayslipPDF(payment, teacher) {
  return new Promise((resolve, reject) => {
    try {
      const dir = path.join(__dirname, '..', 'uploads', 'payslips');
      if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });

      const fileName = `payslip_${payment._id}.pdf`;
      const filePath = path.join(dir, fileName);
      const stream = fs.createWriteStream(filePath);

      const doc = new PDFDocument({ size: 'A4', margin: 50 });
      doc.pipe(stream);

      // Track the cursor ourselves so each section knows where to start.
      // Starts at the top margin (50).
      let y = 50;

      y = drawInfoColumns(doc, y, teacher, payment);
      y = drawEarningsTable(doc, y, payment);
      y = drawTotalsPanel(doc, y, payment);
      if (payment.notes) y = drawNotes(doc, y, payment.notes);
      drawSignatures(doc, y);

      doc.end();
      stream.on('finish', () => resolve(filePath));
      stream.on('error', reject);
    } catch (err) {
      reject(err);
    }
  });
}

/* ------------------------------------------------------- shared helpers */

const eur = (n) => `€ ${Number(n || 0).toFixed(2)}`;
const fmt = (d) => d ? new Date(d).toLocaleDateString('en-GB') : '—';

/* ---------------------------------------------- "Issued to" + "Pay period" */

function drawInfoColumns(doc, startY, teacher, payment) {
  const colW = 240;
  const leftX = 50;
  const rightX = doc.page.width - 50 - colW;
  let y = startY;

  // Column titles + thin underlines
  doc.font('Helvetica-Bold').fontSize(10).fillColor(PALETTE.muted);
  doc.text('ISSUED TO', leftX, y, { lineBreak: false });
  doc.text('PAY PERIOD', rightX, y, { lineBreak: false });

  y += 14;
  doc.strokeColor(PALETTE.divider).lineWidth(0.8)
     .moveTo(leftX, y).lineTo(leftX + colW, y).stroke()
     .moveTo(rightX, y).lineTo(rightX + colW, y).stroke();
  y += 8;

  // Left column — teacher details
  doc.font('Helvetica-Bold').fontSize(12).fillColor(PALETTE.ink)
     .text(`${teacher.firstName} ${teacher.lastName}`, leftX, y, { width: colW });
  doc.font('Helvetica').fontSize(10).fillColor(PALETTE.body)
     .text(teacher.email || '', leftX, doc.y, { width: colW })
     .text(teacher.phone || '', leftX, doc.y, { width: colW });
  const leftBottom = doc.y;

  // Right column — period (manually positioned so both columns share the
  // same starting Y and grow independently without affecting each other)
  let ry = y;
  const labelW = 60;
  drawKV(doc, rightX, ry,        labelW, 'From:',   fmt(payment.periodFrom)); ry += 16;
  drawKV(doc, rightX, ry,        labelW, 'To:',     fmt(payment.periodTo));   ry += 16;
  drawKV(doc, rightX, ry,        labelW, 'Issued:', fmt(payment.createdAt));  ry += 16;
  const rightBottom = ry;

  return Math.max(leftBottom, rightBottom) + 24;
}

function drawKV(doc, x, y, labelW, label, value) {
  doc.font('Helvetica').fontSize(10);
  doc.fillColor(PALETTE.muted).text(label, x, y, { width: labelW, lineBreak: false });
  doc.fillColor(PALETTE.ink).text(String(value), x + labelW, y, { lineBreak: false });
}

/* ---------------------------------------------------------- earnings table */

function drawEarningsTable(doc, startY, payment) {
  const x = 50;
  const w = doc.page.width - 100;
  let y = startY;

  doc.font('Helvetica-Bold').fontSize(10).fillColor(PALETTE.muted)
     .text('EARNINGS', x, y, { lineBreak: false });
  y += 14;

  const cols = [
    { label: 'Description', x: x + 10,  w: 220, align: 'left'  },
    { label: 'Hours',       x: x + 240, w: 70,  align: 'right' },
    { label: 'Rate',        x: x + 320, w: 80,  align: 'right' },
    { label: 'Amount',      x: x + 410, w: 90,  align: 'right' },
  ];

  // Header row
  doc.rect(x, y, w, 22).fill(PALETTE.primary);
  doc.font('Helvetica-Bold').fontSize(10).fillColor(PALETTE.primaryFg);
  for (const c of cols) {
    doc.text(c.label, c.x, y + 6, { width: c.w, align: c.align, lineBreak: false });
  }
  y += 22;

  // Rows: from `breakdown` if present, otherwise a single fallback row.
  const rate = Number(payment.hourlyRate || 0);
  const rows = (Array.isArray(payment.breakdown) && payment.breakdown.length > 0)
    ? payment.breakdown.map((b) => {
        const hourCount = (b.hours || []).length;
        const dateLabel = new Date(b.date).toLocaleDateString('en-GB');
        const hoursText = (b.hours || []).join(', ') || '—';
        return {
          desc: `${dateLabel}  ·  hours: ${hoursText}`,
          hours: hourCount,
          amount: hourCount * rate,
        };
      })
    : [{
        desc: 'Teaching hours',
        hours: Number(payment.hoursWorked || 0),
        amount: Number(payment.grossAmount || 0),
      }];

  const rowH = 22;
  doc.font('Helvetica').fontSize(10);
  rows.forEach((row, i) => {
    if (i % 2 === 0) doc.rect(x, y, w, rowH).fill(PALETTE.zebra);
    doc.fillColor(PALETTE.ink);
    doc.text(row.desc,            cols[0].x, y + 7, { width: cols[0].w, align: cols[0].align, lineBreak: false });
    doc.text(String(row.hours),   cols[1].x, y + 7, { width: cols[1].w, align: cols[1].align, lineBreak: false });
    doc.text(eur(rate),           cols[2].x, y + 7, { width: cols[2].w, align: cols[2].align, lineBreak: false });
    doc.text(eur(row.amount),     cols[3].x, y + 7, { width: cols[3].w, align: cols[3].align, lineBreak: false });
    y += rowH;
  });

  doc.strokeColor(PALETTE.divider).lineWidth(0.8)
     .moveTo(x, y).lineTo(x + w, y).stroke();

  return y + 16;
}

/* ----------------------------------------------------------- totals panel */

function drawTotalsPanel(doc, startY, payment) {
  const w = 200;
  const h = 56;
  const x = doc.page.width - 50 - w;
  const y = startY;

  doc.rect(x, y, w, h).fill(PALETTE.primary);
  doc.font('Helvetica').fontSize(10).fillColor('#FFFFFFCC')
     .text('TOTAL DUE', x + 14, y + 10, { width: w - 28, lineBreak: false });
  doc.font('Helvetica-Bold').fontSize(20).fillColor(PALETTE.primaryFg)
     .text(eur(payment.grossAmount), x + 14, y + 24, { width: w - 28, lineBreak: false });

  // Slim accent stripe on the leading edge.
  doc.rect(x, y, 4, h).fill(PALETTE.accent);

  return y + h + 24;
}

/* ----------------------------------------------------------------- notes */

function drawNotes(doc, startY, notes) {
  const x = 50;
  const w = doc.page.width - 100;
  let y = startY;

  doc.font('Helvetica-Bold').fontSize(10).fillColor(PALETTE.muted)
     .text('NOTES', x, y, { lineBreak: false });
  y += 14;

  doc.font('Helvetica').fontSize(10).fillColor(PALETTE.body);
  doc.text(String(notes), x, y, { width: w });

  return doc.y + 16;
}

/* ------------------------------------------------------------- signatures */

/**
 * Signature block flows naturally after the body content, with a small
 * extra gap. We anchor it to AT LEAST a known Y (so it doesn't sit too
 * close to a short body), but never push it off the page.
 *
 * Two columns side by side: authorised on the left, teacher on the right.
 */
function drawSignatures(doc, afterBodyY) {
  // Sit comfortably below the body. If the body ended early, push the
  // signatures down to ~600 so they don't look orphaned right below the
  // total. If the body ended late, keep the natural position.
  const minY = 600;
  let y = Math.max(afterBodyY + 24, minY);

  // Last-resort safety: never pass the bottom margin. If we'd be too low,
  // pull the signatures up to a sensible bottom-anchored position. This
  // also prevents PDFKit from auto-pagebreaking.
  const bottomCap = doc.page.height - 100;
  if (y > bottomCap) y = bottomCap;

  const colW = 220;
  const left = 50;
  const right = doc.page.width - 50 - colW;
  const lineY = y + 30;

  doc.font('Helvetica').fontSize(10).fillColor(PALETTE.muted);
  doc.text('Authorised signature', left, y,  { width: colW, lineBreak: false });
  doc.text('Teacher signature',    right, y, { width: colW, lineBreak: false });

  doc.strokeColor(PALETTE.ink).lineWidth(0.8)
     .moveTo(left,  lineY).lineTo(left + colW,  lineY).stroke()
     .moveTo(right, lineY).lineTo(right + colW, lineY).stroke();

  doc.fontSize(8).fillColor(PALETTE.muted);
  doc.text('Signature & date', left, lineY + 4, { width: colW, lineBreak: false });
  doc.text('Date: __________________', right, lineY + 4, { width: colW, lineBreak: false });
}

module.exports = generatePayslipPDF;

utils/mail.js
const nodemailer = require('nodemailer');

let transporter = null;
let usingConsole = false;

function getTransporter() {
  if (transporter) return transporter;

  if (!process.env.SMTP_HOST) {
    usingConsole = true;
    transporter = {
      sendMail: async (opts) => {
        // Stand-in: print the email so dev can read it from the server log.
        console.log('\n----- [DEV MAIL] -----');
        console.log('To:      ', opts.to);
        console.log('Subject: ', opts.subject);
        console.log('Text:    \n' + (opts.text || ''));
        console.log('----- [/DEV MAIL] -----\n');
        return { messageId: 'dev-' + Date.now() };
      },
    };
    return transporter;
  }

  transporter = nodemailer.createTransport({
    host: process.env.SMTP_HOST,
    port: Number(process.env.SMTP_PORT || 587),
    secure: String(process.env.SMTP_SECURE || 'false').toLowerCase() === 'true',
    auth: process.env.SMTP_USER
      ? { user: process.env.SMTP_USER, pass: process.env.SMTP_PASS }
      : undefined,
  });
  return transporter;
}

/** Single-language send. Kept for any callers that don't need translations. */
async function sendMail({ to, subject, text, html }) {
  try {
    const tr = getTransporter();
    const from = process.env.MAIL_FROM || 'Salah al-Din <no-reply@school.local>';
    await tr.sendMail({ from, to, subject, text, html });
    return true;
  } catch (err) {
    console.error('Mail send failed:', err.message);
    return false;
  }
}

/**
 * Trilingual send. Caller passes:
 *
 *   {
 *     to: 'parent@example.com',
 *     subject: { ar: '...', en: '...', nl: '...' },
 *     text:    { ar: '...', en: '...', nl: '...' },
 *   }
 *
 * The recipient gets ONE message that contains all three versions, in the
 * order Arabic → English → Dutch, separated by a clear divider line and a
 * language label so the reader can find their language quickly. The Subject
 * line concatenates all three with a separator. If a language is missing
 * from the input, that section is simply skipped.
 *
 * We send to all recipients regardless of which language they prefer — we
 * don't store a language preference per user yet, and being trilingual on
 * outgoing email is what the requirement asks for.
 */
async function sendMailMulti({ to, subject, text }) {
  // The order of the languages in the email body. Arabic first because it's
  // the school's primary language; English second as the lingua-franca
  // fallback; Dutch third for local context.
  const ORDER = [
    { code: 'ar', label: 'العربية' },
    { code: 'en', label: 'English' },
    { code: 'nl', label: 'Nederlands' },
  ];

  const safe = (obj, code) =>
    obj && typeof obj === 'object' ? (obj[code] || '').trim() : (obj || '').toString().trim();

  // Subject: join the three (in the order above), filtering out empty ones.
  const subjectParts = ORDER.map((l) => safe(subject, l.code)).filter(Boolean);
  const finalSubject = subjectParts.join(' · ') || 'Salah al-Din';

  // Body: each section gets a header line with the language label, then
  // a divider between sections.
  const DIVIDER = '\n────────────────────\n';
  const sections = ORDER
    .map((l) => {
      const body = safe(text, l.code);
      if (!body) return null;
      return `[${l.label}]\n${body}`;
    })
    .filter(Boolean);

  const finalText = sections.join(DIVIDER);

  return sendMail({ to, subject: finalSubject, text: finalText });
}

module.exports = {
  sendMail,
  sendMailMulti,
  isUsingConsoleTransport: () => usingConsole,
};
